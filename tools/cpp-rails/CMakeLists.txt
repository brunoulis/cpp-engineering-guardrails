cmake_minimum_required(VERSION 3.20)
include(../../cmake/version.cmake)
project(cpp_rails VERSION ${CPP_RAILS_VERSION} LANGUAGES CXX)

if(NOT DEFINED CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 23)
endif()
if(CMAKE_CXX_STANDARD LESS 23)
  message(FATAL_ERROR "cpp-rails tools require C++23 or newer")
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(cpp_rails_core STATIC src/cpp_rails.cpp src/process.cpp src/reporting.cpp src/baseline.cpp src/quality.cpp src/semantic.cpp src/architecture.cpp src/lint.cpp src/msvc.cpp)
target_compile_features(cpp_rails_core PUBLIC cxx_std_23)
target_compile_definitions(cpp_rails_core PRIVATE CPP_RAILS_VERSION="${CPP_RAILS_VERSION}")
target_include_directories(cpp_rails_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(cpp_rails_core PRIVATE /W4)
else()
  target_compile_options(cpp_rails_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_link_libraries(cpp_rails_core PRIVATE version)
endif()

add_executable(cpp-rails-audit src/main_audit.cpp)
target_link_libraries(cpp-rails-audit PRIVATE cpp_rails_core)
add_executable(cpp-rails-check src/main_check.cpp)
target_link_libraries(cpp-rails-check PRIVATE cpp_rails_core)

include(CTest)
if(BUILD_TESTING)
  add_executable(cpp-rails-selftest tests/selftest.cpp)
  target_link_libraries(cpp-rails-selftest PRIVATE cpp_rails_core)
  add_test(NAME cpp-rails-selftest COMMAND cpp-rails-selftest ${CMAKE_COMMAND})
  add_executable(cpp-rails-roadmap-tests tests/roadmap_tests.cpp)
  target_link_libraries(cpp-rails-roadmap-tests PRIVATE cpp_rails_core)
  add_test(NAME roadmap-regressions COMMAND cpp-rails-roadmap-tests)
  add_executable(cpp-rails-documentation-tests tests/documentation_tests.cpp)
  target_link_libraries(cpp-rails-documentation-tests PRIVATE cpp_rails_core)
  add_test(NAME documentation-consistency COMMAND cpp-rails-documentation-tests ${CMAKE_CURRENT_SOURCE_DIR}/../..)
  add_executable(cpp-rails-scope-tests tests/scope_tests.cpp)
  target_link_libraries(cpp-rails-scope-tests PRIVATE cpp_rails_core)
  add_test(NAME source-scope-regressions COMMAND cpp-rails-scope-tests)
  add_executable(cpp-rails-msvc-tests tests/msvc_tests.cpp)
  target_link_libraries(cpp-rails-msvc-tests PRIVATE cpp_rails_core)
  add_test(NAME msvc-evaluated-context COMMAND cpp-rails-msvc-tests)
  set_tests_properties(msvc-evaluated-context PROPERTIES TIMEOUT 180)
  add_executable(cpp-rails-quality-tests tests/quality_tests.cpp)
  target_link_libraries(cpp-rails-quality-tests PRIVATE cpp_rails_core)
  add_test(NAME cpp-rails-quality COMMAND cpp-rails-quality-tests)
  add_executable(cpp-rails-corpus tests/corpus_runner.cpp)
  target_link_libraries(cpp-rails-corpus PRIVATE cpp_rails_core)
  add_test(NAME corpus-metrics-selftest COMMAND cpp-rails-corpus --self-test)
  add_test(NAME labeled-precision-corpus COMMAND cpp-rails-corpus --check ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/corpus)
  add_executable(cpp-rails-architecture-tests tests/architecture_tests.cpp)
  target_link_libraries(cpp-rails-architecture-tests PRIVATE cpp_rails_core)
  add_test(NAME cpp-rails-architecture COMMAND cpp-rails-architecture-tests ${CMAKE_COMMAND})
  find_program(CPP_RAILS_CLANG NAMES clang++)
  option(CPP_RAILS_REQUIRE_CLANG_TESTS "Require actual Clang semantic tests" OFF)
  if(CPP_RAILS_REQUIRE_CLANG_TESTS AND NOT CPP_RAILS_CLANG)
    message(FATAL_ERROR "CPP_RAILS_REQUIRE_CLANG_TESTS needs a Clang executable")
  endif()
  add_executable(cpp-rails-semantic-tests tests/semantic_tests.cpp)
  target_link_libraries(cpp-rails-semantic-tests PRIVATE cpp_rails_core)
  if(CPP_RAILS_CLANG)
    add_test(NAME cpp-rails-semantic COMMAND cpp-rails-semantic-tests ${CPP_RAILS_CLANG})
  else()
    add_test(NAME cpp-rails-semantic COMMAND cpp-rails-semantic-tests)
    message(STATUS "Clang semantic runtime tests UNVERIFIED; missing-backend protocol tests still run")
  endif()
  set_tests_properties(cpp-rails-semantic PROPERTIES TIMEOUT 240)
  find_program(CPP_RAILS_TIDY NAMES clang-tidy)
  add_executable(cpp-rails-lint-cache-tests tests/lint_cache_tests.cpp)
  target_link_libraries(cpp-rails-lint-cache-tests PRIVATE cpp_rails_core)
  if(CPP_RAILS_CLANG AND CPP_RAILS_TIDY)
    add_test(NAME lint-dependency-cache COMMAND cpp-rails-lint-cache-tests ${CPP_RAILS_CLANG} ${CPP_RAILS_TIDY})
  else()
    add_test(NAME lint-dependency-cache COMMAND cpp-rails-lint-cache-tests)
  endif()
  set_tests_properties(lint-dependency-cache PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180)
  add_executable(cpp-rails-regressions tests/regressions.cpp)
  target_link_libraries(cpp-rails-regressions PRIVATE cpp_rails_core)
  add_test(NAME cpp-rails-regressions COMMAND cpp-rails-regressions)
  set_tests_properties(cpp-rails-regressions PROPERTIES TIMEOUT 180)
  if(WIN32)
    add_executable(cpp-rails-windows-resources tests/windows_resources.cpp)
    target_compile_features(cpp-rails-windows-resources PRIVATE cxx_std_23)
    target_link_libraries(cpp-rails-windows-resources PRIVATE user32 ole32)
    add_test(NAME windows-resource-contracts COMMAND cpp-rails-windows-resources)
  endif()
  add_executable(cpp-rails-fixtures tests/fixture_runner.cpp)
  target_link_libraries(cpp-rails-fixtures PRIVATE cpp_rails_core)
  add_test(NAME compiler-fixtures COMMAND cpp-rails-fixtures ${CMAKE_CXX_COMPILER} ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures)
  add_executable(cpp-rails-safety-examples ../../tests/fixtures/safety_contracts_positive.cpp)
  target_compile_features(cpp-rails-safety-examples PRIVATE cxx_std_17)
  set_target_properties(cpp-rails-safety-examples PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
  add_test(NAME safety-contract-examples COMMAND cpp-rails-safety-examples)
  add_test(NAME cli-audit-smoke COMMAND cpp-rails-audit audit --root ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures --format json)
  add_executable(cpp-rails-example ../../examples/native-guardrails-snippets.cpp)
  target_compile_features(cpp-rails-example PRIVATE cxx_std_20)
  add_test(NAME contract-snippets COMMAND cpp-rails-example)

endif()

install(TARGETS cpp-rails-audit cpp-rails-check RUNTIME DESTINATION bin)
